/// Pair each referral NS name with its glue IP when known.
pub fn format_referral(ns_names: &[String], glue: &HashMap<String, IpAddr>) -> Vec<String> {
    ns_names
        .iter()
        .map(|n| match glue.get(n) {
            Some(ip) => format!("{n} ({ip})"),
            None => n.clone(),
        })
        .collect()
}

`src/dns.rs`:

```rust
pub async fn raw_query_opts(
    server: IpAddr,
    domain: &str,
    rtype: RecordType,
    rd: bool,
) -> Result<(DnsResponse, u128)> {
    use hickory_proto::op::{Message, MessageType, OpCode, Query};
    let name = Name::from_ascii(format!("{domain}."))?;
    let start = Instant::now();
    let resp = tokio::time::timeout(QUERY_TIMEOUT, async {
        let mut client = connect(server).await?;
        let mut msg = Message::query();
        msg.add_query(Query::query(name, rtype))
            .set_message_type(MessageType::Query)
            .set_op_code(OpCode::Query)
            .set_recursion_desired(rd);
        client.send(msg).await.map_err(|e| anyhow!(e))
    })
    .await
    .map_err(|_| anyhow!("timeout"))??;
    Ok((resp, start.elapsed().as_millis()))
}

pub async fn raw_query(server: IpAddr, domain: &str, rtype: RecordType) -> Result<(DnsResponse, u128)> {
    raw_query_opts(server, domain, rtype, true).await
}
```

(hickory 0.25 `ClientHandle::send` takes a message and returns a stream/first response — check the actual API in the vendored source under `~/.cargo/registry` if `send` differs; `DnsHandle::send` yields a stream, take the first item. If assembling via `client.send` proves awkward, an equivalent path is acceptable as long as RD is controllable and the existing `raw_query` behavior is preserved.)
